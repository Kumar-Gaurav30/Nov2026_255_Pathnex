Ansible — Debug the Broken Playbook
Broken
- name "Broken"
  hosts: all
  tasks:
    - name: Install
      yum:
        name: nginx
        state present

Corrected
- name: Broken
  hosts: all
  tasks:
    - name: Install
      yum:
        name: nginx
        state: present

Terraform — Debug the Broken Configuration
Broken
resouce "aws_instance" BadEC2 {
  ami = ami-123
instance_type="c5.large"
tags = {
 Name:"Broken"
}
}

Corrected
resource "aws_instance" "bad_ec2" {
  ami           = "ami-123"
  instance_type = "c5.large"

  tags = {
    Name = "Broken"
  }
}

Kubernetes — Debug the Broken Deployment
Broken
apiVersion apps/v1
kind Deployment
metadata:
name broken
spec:
 replicas 2
 selector:
   matchLabels:
     app broken
 template:
  metadata:
    labels:
      app broken
  spec:
    containers:
     - name app
       image nginx

Corrected
apiVersion: apps/v1
kind: Deployment

metadata:
  name: broken

spec:
  replicas: 2

  selector:
    matchLabels:
      app: broken

  template:
    metadata:
      labels:
        app: broken

    spec:
      containers:
        - name: app
          image: nginx


Jenkins — Matrix Build
Pipeline
pipeline {
    agent any

    environment {
        INSTITUTE_NAME = "Pathnex"
    }

    matrix {
        axes {
            axis {
                name 'JAVA_VERSION'
                values '8', '11', '17'
            }
        }

        stages {
            stage('Build') {
                steps {
                    sh "echo Building with Java $JAVA_VERSION"
                }
            }
        }
    }
}


GitLab CI — Matrix Jobs

stages:
  - build

build:
  stage: build
  image: maven:3.8.1-jdk-17

  parallel:
    matrix:
      - JAVA_VERSION: ['8', '11', '17']

  script:
    - echo "Building with Java $JAVA_VERSION"

Kubernetes
