Jenkins

brew install ansible
ansible --version
chmod 400 ~/.ssh/my-key-pair.pem
ssh -i ~/.ssh/your-key.pem ec2-user@<SERVER_IP>
mkdir -p ~/ansible-nginx && cd ~/ansible-nginx

inventory.ini
[pathnex]
<SERVER_IP> ansible_user=ec2-user ansible_ssh_private_key_file=~/.ssh/your-key.pem
ansible pathnex -i inventory.ini -m ping

nginx.yml

- name: Install Nginx on Pathnex server
  hosts: pathnex
  become: true

  tasks:
    - name: Install nginx
      ansible.builtin.dnf:
        name: nginx
        state: present

    - name: Start and enable nginx
      ansible.builtin.service:
        name: nginx
        state: started
        enabled: true

    - name: Check nginx responds locally
      ansible.builtin.uri:
        url: http://localhost
        status_code: 200
      changed_when: false


ansible-playbook -i inventory.ini nginx.yml --syntax-check
ansible-playbook -i inventory.ini nginx.yml --check --diff

ansible-playbook -i inventory.ini nginx.yml


# Run again: expect changed=0
ansible-playbook -i inventory.ini nginx.yml

# Ad-hoc check of the service
ansible pathnex -i inventory.ini -b -m command -a "systemctl status nginx"


Terraform — EC2 with Key Pair (c6i.8xlarge)

provider "aws" {
  region = "us-east-1"
}

resource "aws_key_pair" "PathnexKey" {
  key_name   = "PathnexKey"
  public_key = "ssh-rsa AAAA...."
}

resource "aws_instance" "PathnexEC2" {
  ami           = "ami-0abcd1234abcd1234"
  instance_type = "c6i.8xlarge"
  key_name      = aws_key_pair.PathnexKey.key_name

  tags = {
    Name = "Pathnex-EC2"
  }
}
